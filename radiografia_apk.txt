=== GITHUB ACTIONS ===
name: Android Build

on:
  push:
    branches:
      - main
      - develop
    tags:
      - "v*.*.*"
  pull_request:
    branches:
      - main
      - develop
  workflow_dispatch:
  release:
    types:
      - created

jobs:
  build:
    name: Build Debug APK
    runs-on: ubuntu-latest

    env:
      API_BASE_URL: ${{ secrets.API_BASE_URL }}
      APP_NAME: ${{ secrets.APP_NAME }}
      BRAND_PRIMARY_COLOR: ${{ secrets.BRAND_PRIMARY_COLOR }}
      BRAND_SECONDARY_COLOR: ${{ secrets.BRAND_SECONDARY_COLOR }}
      SUPPORT_WHATSAPP: ${{ secrets.SUPPORT_WHATSAPP }}
      SUPPORT_EMAIL: ${{ secrets.SUPPORT_EMAIL }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up JDK
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v3

      - name: Grant execute permission to Gradle
        working-directory: android
        run: chmod +x gradlew

      - name: Run Android Lint
        working-directory: android
        run: ./gradlew lintDebug

      - name: Run Unit Tests
        working-directory: android
        run: ./gradlew testDebugUnitTest

      - name: Build Debug APK
        working-directory: android
        run: ./gradlew assembleDebug

      - name: Upload Debug APK
        uses: actions/upload-artifact@v4
        with:
          name: storetd-play-debug-apk
          path: android/app/build/outputs/apk/debug/*.apk

  release:
    name: Build Signed Release APK
    runs-on: ubuntu-latest
    needs: build
    if: startsWith(github.ref, 'refs/tags/v') || github.event_name == 'release' || github.event_name == 'workflow_dispatch'

    env:
      KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
      KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
      KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
      API_BASE_URL: ${{ secrets.API_BASE_URL }}
      APP_NAME: ${{ secrets.APP_NAME }}
      BRAND_PRIMARY_COLOR: ${{ secrets.BRAND_PRIMARY_COLOR }}
      BRAND_SECONDARY_COLOR: ${{ secrets.BRAND_SECONDARY_COLOR }}
      SUPPORT_WHATSAPP: ${{ secrets.SUPPORT_WHATSAPP }}
      SUPPORT_EMAIL: ${{ secrets.SUPPORT_EMAIL }}

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up JDK
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v3

      - name: Grant execute permission to Gradle
        working-directory: android
        run: chmod +x gradlew

      - name: Decode Keystore
        if: env.KEYSTORE_PASSWORD != ''
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
        run: |
          echo "$KEYSTORE_BASE64" | base64 --decode > android/app/release.keystore

      - name: Build Release APK
        if: env.KEYSTORE_PASSWORD != ''
        working-directory: android
        run: |
          sed -i 's/REPLACE_ME_IN_ACTIONS/${{ secrets.TMDB_API_KEY }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          sed -i 's/REPLACE_MAGMA_DEVICE_ID_IN_ACTIONS/${{ secrets.MAGMA_DEVICE_ID }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          sed -i 's/REPLACE_MAGMA_HASH_IN_ACTIONS/${{ secrets.MAGMA_HASH }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          ./gradlew assembleRelease

      - name: Upload Release APK
        if: env.KEYSTORE_PASSWORD != ''
        uses: actions/upload-artifact@v4
        with:
          name: storetd-play-release-apk
          path: android/app/build/outputs/apk/release/*.apk
name: Android Release APK

on:
  workflow_dispatch:
  push:
    tags:
      - "release-*"

jobs:
  build-release:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Set up Java
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: 17

      - name: Set up Gradle
        uses: gradle/actions/setup-gradle@v3

      - name: Decode keystore
        run: |
          printf '%s' "${{ secrets.STORETD_KEYSTORE_BASE64 }}" | tr -d '\r\n ' | base64 -d > storetd-release.jks
          ls -lh storetd-release.jks
          echo "STORETD_KEYSTORE_FILE=${GITHUB_WORKSPACE}/storetd-release.jks" >> "$GITHUB_ENV"

      - name: Make Gradle executable
        run: chmod +x ./gradlew
        working-directory: android

      - name: Build signed release APK
        run: |
          sed -i 's/REPLACE_ME_IN_ACTIONS/${{ secrets.TMDB_API_KEY }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          sed -i 's/REPLACE_MAGMA_DEVICE_ID_IN_ACTIONS/${{ secrets.MAGMA_DEVICE_ID }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          sed -i 's/REPLACE_MAGMA_HASH_IN_ACTIONS/${{ secrets.MAGMA_HASH }}/g' app/src/main/java/com/storetd/play/core/Secrets.kt
          ./gradlew assembleRelease
        working-directory: android
        env:
          STORETD_KEYSTORE_FILE: ${{ github.workspace }}/storetd-release.jks
          STORETD_KEYSTORE_PASSWORD: ${{ secrets.STORETD_KEYSTORE_PASSWORD }}
          STORETD_KEY_ALIAS: ${{ secrets.STORETD_KEY_ALIAS }}
          STORETD_KEY_PASSWORD: ${{ secrets.STORETD_KEY_PASSWORD }}

      - name: Show release outputs
        run: find android/app/build/outputs/apk/release -type f -maxdepth 1 -print

      - name: Upload signed release APK
        uses: actions/upload-artifact@v4
        with:
          name: storetd-play-release-apk
          path: android/app/build/outputs/apk/release/*.apk
